// HairFlow 1.6.0 - long hair groove assistant for ZBrush 2022.0.5 / 2025.0.1
// Configures ZBrush's native Curve stroke so one long strand is laid out as ONE
// curve, reshaped section by section, and committed as ONE continuous stroke.
// No seam, no second stroke start, brush Size / Z Intensity / Alpha untouched.
//
// 1.6.0 (screen-tested in 2025.0.1): in Curve Mode, DRAGGING the curve sculpts on
// every mouse move (stacked rows). Shape Lock turns Zadd/Zsub off while you
// bend the curve; Shape Unlock puts them back exactly; then one click ON the
// curve commits one clean stroke. Curve Step 0.25 beads with DamStandard, so
// Long Strand Curve starts at 0.1.
// ASCII only. Chinese guide: README_zh.md.
//
// 1.5.0: every Stroke path below was resolved by IGetID/IGetTitle in ZBrush
// 2025.0.1 (tests/results/stroke_paths_2025.txt). Do NOT use "Stroke:Curve"
// (that is the sub-palette, not the toggle) or "Stroke:Curve Functions:*"
// (not resolvable by that name; "...:Add" resolves to Stroke:Inventory:Add).
// Numeric variable names carry no underscores (ZBrush 2025 parser issue).
[IConfig, 2021]

[VarDef, hfRMode, 0]
[VarDef, hfRLazy, 1]
[VarDef, hfRRad, 1]
[VarDef, hfRStep, 0.25]
[VarDef, hfRSm, 0]
[VarDef, hfRCStep, 0.25]
[VarDef, hfRSnap, 0]
[VarDef, hfHasSnap, 0]
[VarDef, hfDiagOk, 0]
[VarDef, hfOff, 0]
[VarDef, hfLen, 0]
[VarDef, hfLine, ""]
[VarDef, hfLocked, 0]
[VarDef, hfRZadd, 1]
[VarDef, hfRZsub, 0]

// put Zadd / Zsub back to the values stored by Shape Lock
[RoutineDef, hfUnlock,
  [If, (hfLocked = 1),
    [If, [IExists, "Draw:Zadd"], [ISet, "Draw:Zadd", hfRZadd]]
    [If, [IExists, "Draw:Zsub"], [ISet, "Draw:Zsub", hfRZsub]]
    [VarSet, hfLocked, 0]
  ]
]

// set Curve Step on ZBrush and mirror it on the HairFlow slider
[RoutineDef, hfSetCurveStep,
  [If, [IExists, "Stroke:Curve:Curve Step"], [ISet, "Stroke:Curve:Curve Step", hfVal]]
  [If, [IExists, "ZPlugin:HairFlow:Curve Step"], [ISet, "ZPlugin:HairFlow:Curve Step", hfVal]]
, hfVal]

// append one text line to memory block HFM at hfOff
[RoutineDef, hfDiagLine,
  [VarSet, hfLen, [StrLength, hfLine]]
  [MemWriteString, HFM, hfLine, hfOff, 0]
  [VarSet, hfOff, hfOff + hfLen]
  [MemWriteString, HFM, [StrFromAsc, 13], hfOff, 0]
  [VarSet, hfOff, hfOff + 1]
  [MemWriteString, HFM, [StrFromAsc, 10], hfOff, 0]
  [VarSet, hfOff, hfOff + 1]
]
